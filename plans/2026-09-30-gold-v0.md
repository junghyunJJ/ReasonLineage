````python
def drug_namer(rows: Sequence[dict[str, str]]) -> Callable[[str], set[str]]:
    """A function giving the repoDB drugs, approved for some indication, that a text names as whole words."""
    names = {row["drug_name"].lower(): row["drug_name"] for row in rows if row["status"] == "Approved"}
    pattern = _drug_pattern(names)
    return lambda text: {names[m] for m in pattern.findall((text or "").lower())}


def tested_drug(interventions: Iterable[dict[str, Any]], name_drugs: Callable[[str], set[str]]) -> str | None:
    """The one repoDB drug that every tested intervention names, or None.

    Tested interventions are drugs or biologicals that are not a placebo; one that names no repoDB drug,
    or a second drug, gives None.
    """
    tested = [i.get("name") or "" for i in interventions
              if i.get("type") in TESTED and not PLACEBO.search(i.get("name") or "")]
    drugs = [name_drugs(name) for name in tested]
    named = set().union(*drugs)
    return named.pop() if drugs and all(drugs) and len(named) == 1 else None


def searched_failures(rows: Sequence[dict[str, str]], ind_id: str, records: Iterable[dict[str, Any]]) -> list[dict[str, str]]:
    """Failed drugs from searched trials of one indication (gold design 4.3): the trial stopped for efficacy
    or safety, and its tested drugs are all one repoDB drug that is approved for something, but not for this.
    """
    name_drugs = drug_namer(rows)
    approved = {row["drug_name"] for row in rows if row["ind_id"] == ind_id and row["status"] == "Approved"}
    found = []
    for record in records:
        reason = stop_reason(record["why_stopped"])
        drug = tested_drug(record["interventions"], name_drugs)
        if reason and drug and drug not in approved:
            found.append({"drug": drug, "nct": record["nct"], "status": record["status"].title(),
                          "phase": "/".join(record["phases"]), "reason": reason, "detail": record["why_stopped"],
                          "source": "ctgov"})
    return found
````

````python
"""CT Open as the source of gold v0 (gold v0 design 4): safe loading, trial selection, review, grouping, files."""

import hashlib
import pickle

import pytest

from reasonlineage.gold import ctopen

FIRED = []


def fire():
    FIRED.append("ran")


class Carrier:
    """Loading a pickled Carrier would call fire(): a stand-in for a file that carries code."""

    def __reduce__(self):
        return (fire, ())


PLAIN = pickle.dumps({("NCT00000001", "{}"): {"arms": [], "n": 1.5, "pair": (1, 2)}})


def pin(monkeypatch, sha256):
    monkeypatch.setitem(ctopen.SETS, "test-2025", {"file": "Test.pickle", "commit": "abc123", "sha256": sha256, "after": "2025-01-01"})


def test_plain_data_loads():
    assert ctopen.safe_load(PLAIN) == {("NCT00000001", "{}"): {"arms": [], "n": 1.5, "pair": (1, 2)}}


def test_a_file_that_carries_code_is_refused_without_running_it():
    with pytest.raises(ctopen.SourceError, match="more than plain data"):
        ctopen.safe_load(pickle.dumps({"x": Carrier()}))
    assert FIRED == []


def test_a_pinned_set_is_fetched_once_and_its_checksum_checked(tmp_path, monkeypatch):
    pin(monkeypatch, hashlib.sha256(PLAIN).hexdigest())
    fetched = []

    def download(url):
        fetched.append(url)
        return PLAIN

    assert ctopen.load_set("test-2025", cache=tmp_path, download=download) == ctopen.safe_load(PLAIN)
    assert ctopen.load_set("test-2025", cache=tmp_path, download=lambda url: pytest.fail("fetched twice"))
    assert fetched == ["https://raw.githubusercontent.com/ClinicalTrial-OpenChallenge/CT_Open/abc123/datasets/Test.pickle"]


def test_a_file_that_is_not_the_pinned_one_is_refused(tmp_path, monkeypatch):
    pin(monkeypatch, "0" * 64)
    with pytest.raises(ctopen.SourceError, match="not the pinned"):
        ctopen.load_set("test-2025", cache=tmp_path, download=lambda url: PLAIN)


def test_a_file_that_is_not_a_dict_of_entries_is_refused(tmp_path, monkeypatch):
    listed = pickle.dumps([1, 2])
    pin(monkeypatch, hashlib.sha256(listed).hexdigest())
    with pytest.raises(ctopen.SourceError, match="dict of entries"):
        ctopen.load_set("test-2025", cache=tmp_path, download=lambda url: listed)
````

````python
"""CT Open (arXiv 2604.16742) as the source of gold v0 (gold v0 design).

Every quarter CT Open publishes clinical trial outcome questions whose answers were not public before the
window's cutoff, with the dated reports each answer came from. Gold v0 keeps the primary-endpoint results of
trials that test an already approved drug and were first reported after every panel model's training cutoff.
"""

import hashlib
import io
import pickle
import urllib.request
from pathlib import Path
from typing import Any, Callable

REPO = "ClinicalTrial-OpenChallenge/CT_Open"
COMMIT_2025 = "cc78ff33e90cc8da01f9db4e3b3d35dcc9ce4539"
# Each set's file, pinned by commit and sha256, and the earliest first-report date it accepts (gold v0 design 4.1).
SETS: dict[str, dict[str, str]] = {
    "winter-2025": {"file": "Winter_2025.pickle", "commit": COMMIT_2025,
                    "sha256": "849c36f258a84787a5360500ca246c800880418e5cb1996f73f8cb4927736141", "after": "2025-02-01"},
    "summer-2025": {"file": "Summer_2025.pickle", "commit": COMMIT_2025,
                    "sha256": "1b01f622d68c42d522916734cf4b04fb5a437901b6cea2f7e39df91085ca9567", "after": "2025-09-01"},
}
CACHE = Path.home() / ".cache" / "reasonlineage" / "ctopen"


class SourceError(RuntimeError):
    """A CT Open file could not be fetched, is not the pinned file, or does not hold the expected plain data."""


class _PlainOnly(pickle.Unpickler):
    def find_class(self, module: str, name: str) -> Any:
        raise pickle.UnpicklingError(f"{module}.{name}")


def safe_load(data: bytes) -> Any:
    """The pickled object, if it is plain data (dicts, lists, tuples, strings, numbers).

    A pickle can run code while it loads, so every class and function it asks for is refused.
    """
    try:
        return _PlainOnly(io.BytesIO(data)).load()
    except pickle.UnpicklingError as exc:
        raise SourceError(f"the file holds more than plain data ({exc}); refusing to load it") from None


def _download(url: str) -> bytes:
    with urllib.request.urlopen(url, timeout=120) as response:
        return response.read()


def load_set(name: str, cache: str | Path = CACHE, download: Callable[[str], bytes] = _download) -> dict:
    """The entries of a pinned CT Open set, fetched once into `cache`."""
    if name not in SETS:
        raise SourceError(f"unknown CT Open set {name!r}; add it to ctopen.SETS with its commit, file, sha256 "
                          f"and date (gold v0 design 4.1)")
    pin = SETS[name]
    url = f"https://raw.githubusercontent.com/{REPO}/{pin['commit']}/datasets/{pin['file']}"
    path = Path(cache) / pin["file"]
    if not path.exists():
        try:
            data = download(url)
        except OSError as exc:
            raise SourceError(f"could not fetch {url}: {exc}") from None
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(data)
    data = path.read_bytes()
    digest = hashlib.sha256(data).hexdigest()
    if digest != pin["sha256"]:
        raise SourceError(f"{path} has sha256 {digest}, not the pinned {pin['sha256']}; delete it to fetch again")
    entries = safe_load(data)
    if not isinstance(entries, dict):
        raise SourceError(f"{path} does not hold a dict of entries (gold v0 design 4.2)")
    return entries
````

````python
def first_report(entry: dict[str, Any]) -> str | None:
    """The earliest YYYY-MM-DD date among an entry's reported results, or None."""
    dates = [report.get("date") for report in (entry.get("reported_results") or {}).values() if isinstance(report, dict)]
    dates = [d for d in dates if isinstance(d, str) and DATE.fullmatch(d)]
    return min(dates) if dates else None


def _endpoint(text: Any) -> dict[str, Any] | None:
    """The outcome measure that CT Open stores as dict text, or None when it does not parse."""
    try:
        value = ast.literal_eval(text) if isinstance(text, str) else text
    except (ValueError, SyntaxError):
        return None
    return value if isinstance(value, dict) else None


def _comparison(entry: dict[str, Any]) -> tuple[str, str, str] | None:
    """The one (tested arm, MET or NOT MET, comparator) of an entry, or None."""
    tuples = entry["list_of_tuples"] or []
    if len(tuples) != 1 or len(tuples[0]) != 3 or tuples[0][1] not in ("MET", "NOT MET"):
        return None
    return tuple(tuples[0])


def candidates(entries: dict, after: str, name_drugs: Callable[[str], set[str]]) -> list[dict[str, Any]]:
    """Trials whose primary-endpoint result gold v0 can use (gold v0 design 4.2), in NCT order.

    A trial counts when every primary-endpoint entry holds one comparison and they all agree, its first report
    (the earliest over all its entries, any outcome) is on or after `after`, and its tested arm tests one repoDB
    drug approved for some indication.
    """
    first: dict[str, str] = {}
    primary: dict[str, list[tuple[dict[str, Any], dict[str, Any]]]] = defaultdict(list)
    for (nct, _), entry in entries.items():
        missing = [field for field in FIELDS if field not in entry]
        if missing:
            raise SourceError(f"CT Open entry {nct} has no {', '.join(missing)}; "
                              f"the file's format differs from gold v0 design 4.2")
        date = first_report(entry)
        if date and (nct not in first or date < first[nct]):
            first[nct] = date
        endpoint = _endpoint(entry["endpoint"])
        if endpoint and endpoint.get("outcome_type") == "primaryOutcomes":
            primary[nct].append((endpoint, entry))
    found = []
    for nct in sorted(primary):
        if nct not in first or first[nct] < after:
            continue
        comparisons = [_comparison(entry) for _, entry in primary[nct]]
        if None in comparisons or len({c[1] for c in comparisons}) != 1:
            continue
        endpoint, entry = min(primary[nct], key=lambda pair: str(sorted(pair[0].items())))
        arm_label, label, comparator = _comparison(entry)
        arm = next((a for a in entry["arms"] if a.get("label") == arm_label), None)
        drug = repodb.tested_drug(arm.get("interventions") or [], name_drugs) if arm else None
        if drug is None:
            continue
        found.append({"nct": nct, "title": entry.get("brief_title"), "drug": drug, "arm": arm_label,
                      "comparator": comparator, "endpoint": {"measure": endpoint.get("measure"),
                                                            "timeFrame": endpoint.get("timeFrame")},
                      "label": label, "first_report": first[nct]})
    return found


def attach(found: list[dict[str, Any]], records: dict[str, dict[str, Any]]) -> list[dict[str, Any]]:
    """The candidates with each trial's conditions and phases from its ClinicalTrials.gov record.

    A record that could not be fetched leaves both empty and is marked, so the review can judge it.
    """
    out = []
    for candidate in found:
        record = records.get(candidate["nct"], {})
        out.append({**candidate, "conditions": record.get("conditions") or [], "phases": record.get("phases") or [],
                    **({"fetch_error": record["error"]} if "error" in record else {})})
    return out
````

````python
def check_review(review: dict[str, Any], found: list[dict[str, Any]]) -> list[str]:
    """What breaks the rules of gold v0 design 4.3 in the review record; empty when usable."""
    known = {candidate["nct"] for candidate in found}
    problems: list[str] = []
    seen: set[str] = set()
    for kind in ("keep", "drop"):
        for item in review.get(kind, []):
            nct = item.get("nct")
            if nct not in known:
                problems.append(f"{nct}: not among the candidates")
            elif nct in seen:
                problems.append(f"{nct}: reviewed twice")
            seen.add(nct)
            if not item.get("why"):
                problems.append(f"{nct}: needs a reason ('why')")
            if kind == "keep" and not item.get("display_name"):
                problems.append(f"{nct}: a kept trial needs a display_name")
    problems += [f"{nct}: not reviewed; keep or drop it" for nct in sorted(known - seen)]
    return problems


def group(kept: list[dict[str, Any]], set_name: str) -> list[tuple[dict[str, Any], ...]]:
    """(MET, NOT MET, NOT MET) groups with three different diseases, in an order fixed by the set name.

    MET and NOT MET trials are each sorted by NCT and shuffled by a seed from the set name; each MET trial
    takes the first two unused NOT MET trials whose diseases differ from its own and from each other.
    """
    met = {t["nct"]: t for t in kept if t["label"] == "MET"}
    not_met = {t["nct"]: t for t in kept if t["label"] == "NOT MET"}
    pool = shuffled(sorted(not_met), f"ctopen-{set_name}-not-met")
    used: set[str] = set()
    groups = []
    for nct in shuffled(sorted(met), f"ctopen-{set_name}-met"):
        diseases = {met[nct]["display_name"]}
        picks = []
        for other in pool:
            if other not in used and not_met[other]["display_name"] not in diseases:
                picks.append(other)
                diseases.add(not_met[other]["display_name"])
                if len(picks) == 2:
                    break
        if len(picks) == 2:
            used.update(picks)
            groups.append((met[nct], not_met[picks[0]], not_met[picks[1]]))
    return groups


def _phase(phases: list[str]) -> str:
    names = [p.replace("EARLY_PHASE", "early phase ").replace("PHASE", "phase ") for p in phases if p != "NA"]
    return "/".join(names) or "phase not reported"


def build(found: list[dict[str, Any]], review: dict[str, Any], set_name: str, tasks_dir: str | Path,
          gold_path: str | Path, items_dir: str | Path) -> list[Path]:
    """Write one task file per group, the gold file, and one items file per task (gold v0 design 4.4).

    Nothing is written when the review breaks a rule or the task folder holds files this review does not build.
    """
    problems = check_review(review, found)
    if problems:
        raise ReviewError("; ".join(problems))
    by_nct = {candidate["nct"]: candidate for candidate in found}
    kept = [by_nct[item["nct"]] | {"display_name": item["display_name"]} for item in review.get("keep", [])]
    groups = group(kept, set_name)
    ids = [f"ctopen-{set_name}-{i:02d}" for i in range(1, len(groups) + 1)]
    tasks_dir, gold_path, items_dir = Path(tasks_dir), Path(gold_path), Path(items_dir)
    stale = [f"{path} is not built by this review; delete it, then build again"
             for path in sorted(tasks_dir.glob("*.json")) if path.stem not in ids]
    if stale:
        raise ReviewError("; ".join(stale))
    pin = SETS[set_name]
    gold = {"source": {"repo": REPO, "commit": pin["commit"], "path": f"datasets/{pin['file']}", "sha256": pin["sha256"],
                       "after": pin["after"]},
            "tasks": {}}
    tasks_dir.mkdir(parents=True, exist_ok=True)
    items_dir.mkdir(parents=True, exist_ok=True)
    written = []
    for task_id, trials in zip(ids, groups):
        names = {f"{t['drug']} for {t['display_name']}": t for t in trials}
        order = shuffled(list(names), task_id)
        lines = [f"- {name}: {names[name]['arm']} vs {names[name]['comparator']}, {_phase(names[name]['phases'])}; "
                 f"primary endpoint: {names[name]['endpoint']['measure']} ({names[name]['endpoint']['timeFrame']})"
                 for name in order]
        body = {"id": task_id, "question": QUESTION + "\n" + "\n".join(lines), "candidates": order,
                "category": "biomedical", "data_sensitivity": "public"}
        path = tasks_dir / f"{task_id}.json"
        path.write_text(json.dumps(body, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
        written.append(path)
        met, *failed = list(names)
        gold["tasks"][task_id] = {"approved": [met], "failed": sorted(failed)}
        items = {"items": [{"item": name, "outcome_date": t["first_report"]} for name, t in names.items()]}
        (items_dir / f"{task_id}.json").write_text(json.dumps(items, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    gold_path.parent.mkdir(parents=True, exist_ok=True)
    gold_path.write_text(json.dumps(gold, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    return written
````

````python
#!/usr/bin/env python3
"""Build a gold v0 set from CT Open (gold v0 design 4).

    python scripts/build_ctopen.py candidates SET   # gold/ctopen-SET/candidates.json and trials.json
    (Claude writes gold/ctopen-SET/review.json by the rules in gold v0 design 4.3)
    python scripts/build_ctopen.py build SET        # tasks/ctopen-SET/*.json, gold.json and items/*.json
"""

import argparse
import json
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))  # this checkout's code, not an installed copy

from reasonlineage.gold import ctopen, repodb, trials  # noqa: E402


def cmd_candidates(args: argparse.Namespace) -> int:
    try:
        entries = ctopen.load_set(args.set, cache=args.cache)
        rows = repodb.load_rows(cache=args.repodb_cache)
        found = ctopen.candidates(entries, ctopen.SETS[args.set]["after"], repodb.drug_namer(rows))
    except (ctopen.SourceError, repodb.SourceError) as exc:
        print(f"cannot read the sources: {exc}", file=sys.stderr)
        return 1
    args.gold_dir.mkdir(parents=True, exist_ok=True)
    records = trials.fetch_trials([c["nct"] for c in found], args.gold_dir / "trials.json")
    found = ctopen.attach(found, records)
    (args.gold_dir / "candidates.json").write_text(json.dumps(found, ensure_ascii=False, indent=1) + "\n", encoding="utf-8")
    met = sum(c["label"] == "MET" for c in found)
    missing = sorted(c["nct"] for c in found if "fetch_error" in c)
    print(f"{len(found)} candidate trials ({met} MET, {len(found) - met} NOT MET)"
          + (f"; no ClinicalTrials.gov record: {', '.join(missing)}" if missing else ""))
    return 0


def cmd_build(args: argparse.Namespace) -> int:
    found = json.loads((args.gold_dir / "candidates.json").read_text(encoding="utf-8"))
    review = json.loads((args.gold_dir / "review.json").read_text(encoding="utf-8"))
    try:
        written = ctopen.build(found, review, args.set, args.tasks_dir, args.gold_dir / "gold.json", args.gold_dir / "items")
    except ctopen.ReviewError as exc:
        print(f"review.json breaks the rules: {exc}", file=sys.stderr)
        return 1
    print(f"{len(written)} task files in {args.tasks_dir}; gold file {args.gold_dir / 'gold.json'}")
    return 0


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("step", choices=["candidates", "build"])
    parser.add_argument("set", choices=sorted(ctopen.SETS), help="a pinned CT Open set (gold v0 design 4.1)")
    parser.add_argument("--cache", type=Path, default=ctopen.CACHE, help="where CT Open files are kept (outside the repo)")
    parser.add_argument("--repodb-cache", type=Path, default=repodb.CACHE, help="where the repoDB file is kept")
    parser.add_argument("--gold-dir", type=Path, help="default: gold/ctopen-SET")
    parser.add_argument("--tasks-dir", type=Path, help="default: tasks/ctopen-SET")
    args = parser.parse_args(argv)
    args.gold_dir = args.gold_dir or ROOT / "gold" / f"ctopen-{args.set}"
    args.tasks_dir = args.tasks_dir or ROOT / "tasks" / f"ctopen-{args.set}"
    return {"candidates": cmd_candidates, "build": cmd_build}[args.step](args)


if __name__ == "__main__":
    sys.exit(main())
````

````bash
#!/usr/bin/env bash
# Level-1 measurement of a gold set (gold design 7, gold v0 design 4.5): generation 0 only, one task at
# a time, then the gold summary. A task that already has a finished run in OUT is skipped. A task
# whose run stopped halfway stops the script: resume that run first, then run the script again.
#
#   scripts/run_gold.sh SET [OUT]      # SET: a folder under tasks/ and gold/, e.g. repodb-dev; default OUT: runs/gold-SET
set -euo pipefail
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
SET="${1:?usage: scripts/run_gold.sh SET [OUT]}"
OUT="${2:-$ROOT/runs/gold-$SET}"
export PYTHONPATH="$ROOT/src"   # this checkout's code, not an installed copy
if ! compgen -G "$ROOT/tasks/$SET/*.json" > /dev/null; then
  echo "no task files in tasks/$SET; build the set first" >&2
  exit 1
fi
mkdir -p "$OUT"
for task in "$ROOT"/tasks/"$SET"/*.json; do
  id="$(basename "$task" .json)"
  if compgen -G "$OUT/*-$id/result.json" > /dev/null; then
    echo "skip $id: already finished"
    continue
  fi
  if compgen -G "$OUT/*-$id/header.json" > /dev/null; then
    echo "stop: $id has a run in $OUT that did not finish; resume it with" \
         "'PYTHONPATH=$ROOT/src python -m reasonlineage resume <run folder>', then run this script again" >&2
    exit 1
  fi
  python -m reasonlineage run "$task" --generations 0 --seed 1 --out "$OUT"
done
python -m reasonlineage gold --gold "$ROOT/gold/$SET/gold.json" "$OUT"/*/ --out "$OUT/report.json"
````

3. 남기는 임상에는 `display_name`을 적는다. 질문에 들어갈 영어 소문자 병명이고, 같은 병은 같은 이름으로 적는다(묶을 때 이 이름으로 병이 겹치는지 본다). CT.gov 질병 이름이 집단을 가리키면(예: `kidney transplant recipients`) 실제로 치료하려는 병을 적는다.
4. `why`는 한 줄 영어로 쓴다. 남기면 무엇이 시험 약이고 왜 새 적응증인지, 빼면 어느 조건에 걸리는지 쓴다.
5. CT.gov 기록이 없는 임상(`NO TRIAL RECORD`)은 제목과 시험 군 정보로 판단하고, 판단할 수 없으면 뺀다.

형식 (설계 4.3):

````json
{"keep": [{"nct": "NCT…", "display_name": "…", "why": "…"}],
 "drop": [{"nct": "NCT…", "why": "…"}]}
````
